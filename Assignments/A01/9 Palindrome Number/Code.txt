class Solution 
{
    public:
    bool isPalindrome(int x) 
    {
        if (x<0)
        {
            return false;
        }
      long r=0;
      int temp=x;
        while(x>0)
        {
            int i=x%10;
            r=r*10+i;
            x=x/10;
        }
        if (r==temp) 
        {
            return true;
        }
        else
        {
            return false;
        }
        
        
    }
};
