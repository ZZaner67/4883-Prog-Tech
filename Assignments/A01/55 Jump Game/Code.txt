class Solution 
{
public:
    bool canJump(vector<int>& nums) 
    {
        int maxm = 0; // Variable to track the farthest index we can reach
        for (int i = 0; i < nums.size(); i++) 
        {
            if (i <= maxm) // Check if the current index is reachable
            {
                maxm = max(maxm, i + nums[i]); // Update the farthest reach
            } else 
            {
                return false; // If current index is not reachable, return false
            }
        }
        return true; // If we traverse successfully, return true
    }
};
