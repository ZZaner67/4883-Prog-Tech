class Solution 
{
    public:
    int hammingWeight(int n) 
    {
        int x=n;
        int count=0;
        while (x >= 1) 
        {
            if (x % 2 == 0) 
            {
                x = x/2;
            }
            else 
            {
                count++;
                x = (x-1)/2;
            }
        }
        return count;
    }
};
