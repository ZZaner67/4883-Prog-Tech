class Solution 
{
public:
    int longestConsecutive(vector<int>& nums) 
    {
        if( nums.size()==0)
        {
            return 0;
        }
        sort(nums.begin() , nums.end());

        int c=1 , k=1;
        for( int i=1 ; i<nums.size() ; i++ )
        {
            if( nums[i]==nums[i-1]+1 )
            {
                k++;
            }
            else if( nums[i]!=nums[i-1] )
            {
                if( c<k )
                {
                    c=k;
                }
                k=1;
            }
        }
        
        if( c<k )
        {
            c=k;
        }
        return c;
    }
};
