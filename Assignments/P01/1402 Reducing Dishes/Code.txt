class Solution {
public:
    int maxSatisfaction(vector<int>& satisfaction) 
    {
        int size = satisfaction.size();
        //Sorts the array in decreasing order
        sort(satisfaction.begin(), satisfaction.end(), greater<int>());
        //Initialize Variables
        int curSum = 0;
        int satis = 0;
        /*The Satisfaction loop that depends on the number of variables
        in the array*/
        for(int i=0; i<size; i++)
        {
            //Compares the NEGATIVE of the current value to the curSum
            if(-satisfaction[i] > curSum)
                return satis;
            curSum += satisfaction[i];
            satis += curSum;
        }

        return satis;
    }
};
